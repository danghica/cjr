Fixpoint live_identifiers (words : list Z) (count : nat) : list Z :=
 match count with
 | O => []
 | S i => let ids := live_identifiers words i in
   let id := slot_identifier words i in
   if bool_decide (1 <= id) then ids ++ [id] else ids
 end.
