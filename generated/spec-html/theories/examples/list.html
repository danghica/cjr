Fixpoint is_list (xs : list Z) (p : val_cjr) : iProp Σ :=
    match xs with
    | [] =>
        ∃ o, ⌜ p = LitV (LitObj o) ⌝ ∗
          ObjId o ↦ₒ[0] LitV (LitBool false) ∗
          ObjId o ↦ₒ[1] LitV (LitInt 0) ∗
          ObjId o ↦ₒ[2] LitV (LitObj o)
    | x :: xs' =>
        ∃ o nxt, ⌜ p = LitV (LitObj o) ⌝ ∗
          ObjId o ↦ₒ[0] LitV (LitBool true) ∗
          ObjId o ↦ₒ[1] LitV (LitInt x) ∗
          ObjId o ↦ₒ[2] nxt ∗
          is_list xs' nxt
    end.
