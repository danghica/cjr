lemma swap_wp:
    typeclass maximal implicit !(_, type(cjrGS(Σ)))
    typeclass maximal implicit !(_, type(invGS_gen(HasLc, Σ)))
    parameter(l1, l2, type(Z))
    parameter(v1, type(_ (inferred)))
    parameter(v2, type(_ (inferred)))
    statement:
        consuming_implies:
            owns_raw(RawId(l1), v1)
            consuming_implies:
                owns_raw(RawId(l2), v2)
                wp_partial:
                    swap_at(l1, l2)
                    post binder(_)
                    postcondition:
                        separately:
                            owns_raw(RawId(l1), v2)
                            owns_raw(RawId(l2), v1)
Original declaration (proof-free)
Lemma swap_wp `{!cjrGS Σ} `{!invGS_gen HasLc Σ} (l1 l2 : Z) v1 v2 :
  RawId l1 ↦ᵣ v1 -∗ RawId l2 ↦ᵣ v2 -∗
  WP swap_at l1 l2 {{ _, RawId l1 ↦ᵣ v2 ∗ RawId l2 ↦ᵣ v1 }}.
