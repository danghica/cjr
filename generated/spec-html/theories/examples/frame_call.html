lemma write_call_wp:
    typeclass maximal implicit !(_, type(cjrGS(Σ)))
    typeclass maximal implicit !(_, type(invGS_gen(HasLc, Σ)))
    parameter(p, q, type(Z))
    parameter(v, type(_ (inferred)))
    statement:
        consuming_implies:
            owns_raw(RawId(p), v)
            consuming_implies:
                owns_raw(RawId(q), LitV(LitInt(3)))
                wp_partial:
                    write_call(p, q)
                    post binder(w)
                    postcondition:
                        separately:
                            pure(equals(w, LitV(LitInt(3))))
                            separately:
                                owns_raw(RawId(p), LitV(LitInt(9)))
                                owns_raw(RawId(q), LitV(LitInt(3)))
