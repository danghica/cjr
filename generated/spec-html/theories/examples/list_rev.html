        typeclass maximal implicit !(_, type(invGS_gen(HasLc, Σ)))

    statement:
        arrow:
            equals(append(rev(u), cons(x, v)), xs)
            consuming_implies:
                owns_stack(StackId(la), acc)
                consuming_implies:
                    owns_stack(StackId(lc), cur)
                    consuming_implies:
                        is_list(cons(x, v), cur)
                        consuming_implies:
                            is_list(u, acc)
                            wp_partial:
                                rev_step_at(la, lc)
                                post binder(_)
                                postcondition:
                                    exists:
                                        binders:
                                            parameter(nxt, type(_ (inferred)))
                                        separately:
                                            pure(equals(append(rev(cons(x, u)), v), xs))
                                            separately:
                                                owns_stack(StackId(la), cur)
                                                separately:
                                                    owns_stack(StackId(lc), nxt)
                                                    separately:
                                                        is_list(v, nxt)
                                                        is_list(cons(x, u), cur)
Original declaration (proof-free)
Lemma wp_rev_step_at (xs u : list Z) (x : Z) (v : list Z)
      (la lc : Z) (acc cur : val_cjr) :
    rev u ++ x :: v = xs →
    StackId la ↦ₛ acc -∗ StackId lc ↦ₛ cur -∗
    is_list (x :: v) cur -∗ is_list u acc -∗
    WP (rev_step_at la lc)
      {{ _, ∃ nxt,
         ⌜ rev (x :: u) ++ v = xs ⌝ ∗
         StackId la ↦ₛ cur ∗ StackId lc ↦ₛ nxt ∗
         is_list v nxt ∗ is_list (x :: u) cur }}.
