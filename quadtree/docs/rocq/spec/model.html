build_tree

cjr.quadtree.spec.model.build_tree

recursive function build_tree:
    parameter(d, type(nat))
    parameter(b, type(rectangle))
    parameter(xs, type(list(point)))
    type:
        qtree
    body:
        match RegularStyle:
            xs
            case:
                patterns:
                    notation "[ ]"
                Empty
            case:
                patterns(cons(p, rest))
                match RegularStyle:
                    rest
                    case:
                        patterns:
                            notation "[ ]"
                        Leaf(p)
                    case:
                        patterns(cons(_, _))
                        match RegularStyle:
                            d
                            case:
                                patterns(scope nat(0))
                                Leaf(p)
                            case:
                                patterns(S(n))
                                Branch:
                                    build_tree:
                                        n
                                        child_box(n, b, LL)
                                        select_points(child_box(n, b, LL), xs)
                                    build_tree:
                                        n
                                        child_box(n, b, LR)
                                        select_points(child_box(n, b, LR), xs)
                                    build_tree:
                                        n
                                        child_box(n, b, UL)
                                        select_points(child_box(n, b, UL), xs)
                                    build_tree:
                                        n
                                        child_box(n, b, UR)
                                        select_points(child_box(n, b, UR), xs)
Original declaration (proof-free)
Fixpoint build_tree (d : nat) (b : rectangle) (xs : list point) : qtree :=
  match xs with
  | [] => Empty
  | p :: rest => match rest with
      | [] => Leaf p
      | _ :: _ => match d with
          | 0%nat => Leaf p
          | S n => Branch
              (build_tree n (child_box n b LL) (select_points (child_box n b LL) xs))
              (build_tree n (child_box n b LR) (select_points (child_box n b LR) xs))
              (build_tree n (child_box n b UL) (select_points (child_box n b UL) xs))
              (build_tree n (child_box n b UR) (select_points (child_box n b UR) xs))
          end
      end
  end.
